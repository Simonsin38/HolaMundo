# Comandos de clase

pwd: Directorio Actual  
ls: Listar archivos del directorio  
ls -al: Listar archivos del directorio incluido archivos ocultos.  
cd: Cambiar de directorio  
cd ..: Salir del directorio  
mkdir: Crear nuevo directorio  
clear: Limpiar pantalla  
git config --list: Lista de configuraciones git  
touch: Crear archivo  
vim: Editar archivos  
q: Salir sin guardar  
x: Salir y guardar  
git init: Crear repositorio  
git config --global user.name "Nombre y Apellido": Cambiar el nombre de usuario de git  
git config --global user.email "Email": Correo electronico del usuario en git  
git add: Añadir archivo al stage  
git commit -m: Hacer commit a los archivos en el stage
